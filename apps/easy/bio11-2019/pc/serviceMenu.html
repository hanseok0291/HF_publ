function checkBankTime(){
var formData = $("#frm").serialize();
$.ajax(
{
type: "post",
url: ""+URL_PATH+"?_method=checkBankTime4Pop",
dataType : "json",
data: formData,
success:function(data){
if(data.code == "9999"){
alert("은행 시스템 작업으로 삭제가 불가 합니다.");
return false;
}else if(data.code =="0000"){
delAcctno();
}else if(data.code =="session"){
$("#_method").val("goErrorPage");
$("#frm").attr('action',''+URL_PATH).submit();
}else{
alert(data.msg);
return false;
}
},
error:function(e){
alert("서비스 오류가 발생되었습니다.\n다시 시도 하십시오.", function(){
$("#PBnkCd").val("");
});
return false;
}
});
}

$("#frm").attr('action',''+URL_PATH).submit();
}else {
alert(data.msg);
}
},
error:function(e){
alert("서비스 오류가 발생되었습니다.\n다시 시도 하십시오.");
return false;
},
complete:function(){
$("#PBnkCd").val("");
}
});
}

function acctNoCheck(){

var formData = $("#frm").serialize();

{
type: "post",
url: ""+URL_PATH+"?_method=acctInfoCheck",
dataType : "json",
data:formData,

success:function(data){

result = true;
}else{
result = false;
}
},
error:function(e){
result = false;
}
});
return result;
}
});
